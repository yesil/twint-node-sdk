Dark Theme Example

Markup

<div style="background: #1a1a1a; padding: 20px; border-radius: 8px;">
  <pay-with-twint 
    id="dark-payment"
    theme="dark"
    success
    api-url="/twint"
    reference="DARK-2024-003"
    amount="199.00"
    merchant-name="Night Shop"
    logo-url="https://www.twint.ch/content/themes/twint/images/twint-logo.svg">
  </pay-with-twint>
  
  <script>
    // Dark theme with event handling
    const darkPayment = document.getElementById('dark-payment');
    
    darkPayment.addEventListener('payment-started', (e) => {
      console.log('Dark theme payment started', e.detail);
    });
    
    // Custom dark theme notification
    darkPayment.addEventListener('payment-completed', (e) => {
      setTimeout(() => {
        document.body.style.background = '';
      }, 500);
    });
  </script>
</div>
